run_all_experiments_v2.sh:

#!/usr/bin/bash
# =============================================================================
# run_all_experiments_v2.sh
# Automated workflow for AMF re-allocation experiments (7 modes).
#
# New in v2:
#   06_split_gnb_simultaneous : AMF2 switch + gNB2 start + restart second-half UEs
#                               (all triggered together)
#   07_split_gnb_sequential   : AMF2 switch first, wait for recovery, then
#                               gNB2 start + restart second-half UEs
#
# The orchestrator computes new metrics (rrc_latency_ms, post_restart_latency_ms)
# automatically; this script only drives the sequencing.
# =============================================================================

set -uo pipefail

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------
ORCHESTRATOR="python3 sctp_proxy/amf_orchestrator_v5.py"
UE_POOL="ue1,ue2,ue3,ue4,ue5,ue6"
GNB_POOL_SINGLE="gnb1"
GNB_POOL_BOTH="gnb1,gnb2"
WAIT_BEFORE=10
WAIT_AFTER=60
COOLDOWN=30

# ---------------------------------------------------------------------------
# Determine output root directory
# ---------------------------------------------------------------------------
TIMESTAMP=$(date +"%Y.%m.%d_%H-%M-%S")
CUSTOM_LABEL="${1:-}"

if [ -n "$CUSTOM_LABEL" ]; then
    ROOT_DIR="results/${TIMESTAMP}_${CUSTOM_LABEL}"
else
    ROOT_DIR="results/${TIMESTAMP}"
fi
mkdir -p "$ROOT_DIR"

# Subdirectories for each experiment
DIR_SEQ="$ROOT_DIR/01_sequential"
DIR_CONC="$ROOT_DIR/02_concurrent"
DIR_BF="$ROOT_DIR/03_backforth"
DIR_PART="$ROOT_DIR/04_partial"
DIR_GNB="$ROOT_DIR/05_isolate_gnb"
DIR_SPLIT_SIM="$ROOT_DIR/06_split_gnb_simultaneous"
DIR_SPLIT_SEQ="$ROOT_DIR/07_split_gnb_sequential"

for d in "$DIR_SEQ" "$DIR_CONC" "$DIR_BF" "$DIR_PART" "$DIR_GNB" "$DIR_SPLIT_SIM" "$DIR_SPLIT_SEQ"; do
    mkdir -p "$d"
done

# ---------------------------------------------------------------------------
# Helper functions
# ---------------------------------------------------------------------------

ensure_amf1() {
    echo "[CHECK] Ensuring proxy is on AMF1..."
    STATUS=$($ORCHESTRATOR --command status 2>/dev/null || echo "ERROR")
    if echo "$STATUS" | grep -q "Backend: amf1"; then
        echo "[OK] Proxy already on AMF1"
        return 0
    fi
    echo "[INFO] Switching proxy back to AMF1..."
    $ORCHESTRATOR --command switch --target amf1
    sleep 3
}

ensure_gnb2_stopped() {
    echo "[CHECK] Ensuring gnb2 is stopped for clean baseline..."
    if sudo docker ps --format "{{.Names}}" | grep -q "^gnb2$"; then
        echo "[INFO] Stopping gnb2..."
        sudo docker stop gnb2 >/dev/null 2>&1
        sleep 2
    fi
    echo "[OK] gnb2 is not running"
}

run_experiment() {
    local name="$1"
    local cmd="$2"
    local outdir="$3"
    local extra_args="${4:-}"

    echo ""
    echo "======================================================================"
    echo "  EXPERIMENT: $name"
    echo "  Output: $outdir"
    echo "======================================================================"

    # Experiments 1–5 and 6–7 all need AMF1 baseline, except back-forth
    if [ "$name" != "backforth" ]; then
        ensure_amf1
    fi

    # Split-gNB experiments need a clean gnb2 state before they begin
    if [[ "$name" == split_gnb* ]]; then
        ensure_gnb2_stopped
    fi

    echo "[WAIT] Stabilizing for ${WAIT_BEFORE}s before $name..."
    sleep "$WAIT_BEFORE"

    eval "$cmd $extra_args --output-dir \"$outdir\" --basename amf_realloc"

    if [ $? -eq 0 ]; then
        echo "[DONE] $name completed. Results in: $outdir"
    else
        echo "[WARN] $name may have encountered an error. Check logs above."
    fi

    echo "[WAIT] Cooldown ${COOLDOWN}s before next experiment..."
    sleep "$COOLDOWN"
}

# ---------------------------------------------------------------------------
# Main workflow
# ---------------------------------------------------------------------------

echo "======================================================================"
echo "  AMF Re-allocation Experiment Workflow v2"
echo "  Root output: $ROOT_DIR"
echo "======================================================================"

# Verify orchestrator exists
if [ ! -f "sctp_proxy/amf_orchestrator_v5.py" ]; then
    echo "[ERROR] sctp_proxy/amf_orchestrator_v5.py not found!"
    echo "        Please ensure the updated orchestrator is in place."
    exit 1
fi

# Verify proxy is reachable
STATUS=$($ORCHESTRATOR --command status 2>/dev/null || echo "ERROR")
if echo "$STATUS" | grep -q "ERROR"; then
    echo "[ERROR] Cannot reach sctp-proxy. Is it running?"
    exit 1
fi

# ---------------------------------------------------------------------------
# 1) SEQUENTIAL (stagger = 5 s)
# ---------------------------------------------------------------------------
run_experiment "SEQUENTIAL (stagger=5s)" \
    "$ORCHESTRATOR --command sequential --target amf2 --wait-before $WAIT_BEFORE --wait-after $WAIT_AFTER --stagger 5 --ue-pool $UE_POOL --gnb-pool $GNB_POOL_SINGLE" \
    "$DIR_SEQ"

# ---------------------------------------------------------------------------
# 2) CONCURRENT

def send_proxy_cmd(cmd: str, host: str, port: int, timeout: float = 5.0) -> str:
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.settimeout(timeout)
    try:
        sock.sendto(cmd.encode(), (host, port))
        data, _ = sock.recvfrom(4096)
        return data.decode().strip()
    except socket.timeout:
        return "ERROR: Timeout"
    finally:
        sock.close()


def get_proxy_status(host: str, port: int) -> str:
    return send_proxy_cmd("STATUS", host, port)

class LogParser:
    UE_TS_RE = re.compile(r"\[(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3})\]")
    UE_PATTERNS = {
        "mm_deregistered": re.compile(r"UE switches to state \[MM-DEREGISTERED"),
        "mm_registered": re.compile(r"UE switches to state \[MM-REGISTERED/NORMAL-SERVICE\]"),
        "initial_registration": re.compile(r"Sending Initial Registration"),
        "registration_success": re.compile(r"Initial Registration is successful"),
        "pdu_session_success": re.compile(r"PDU Session establishment is successful"),
        "tun_up": re.compile(r"Connection setup for PDU session\[.*\] is successful, TUN interface\[.*, (\d+\.\d+\.\d+\.\d+)\]"),
        "cell_selection_fail": re.compile(r"Cell selection failure"),
        "rrc_connected": re.compile(r"UE switches to state \[RRC-CONNECTED\]"),
        "cm_connected": re.compile(r"UE switches to state \[CM-CONNECTED\]"),
        "registration_reject": re.compile(r"Registration Reject|MM status received with cause"),
    }

    AMF_TS_RE = re.compile(r"(\d{2}/\d{2} \d{2}:\d{2}:\d{2}\.\d{3}):")
    AMF_PATTERNS = {
        "initial_ue_msg": re.compile(r"InitialUEMessage"),
        "registration_complete": re.compile(r"Registration complete"),
        "release_sm_context": re.compile(r"Release SM context"),
        "amf_session_added": re.compile(r"Added\] Number of AMF-Sessions is now (\d+)"),
        "amf_session_removed": re.compile(r"Removed\] Number of AMF-Sessions is now (\d+)"),
        "gnb_ue_added": re.compile(r"Added\] Number of gNB-UEs is now (\d+)"),
        "gnb_ue_removed": re.compile(r"Removed\] Number of gNB-UEs is now (\d+)"),
        "known_ue": re.compile(r"known UE by"),
        "holding_ng": re.compile(r"Holding NG Context"),
        "clear_ng": re.compile(r"Clear NG Context"),
    }

    GNB_TS_RE = re.compile(r"\[(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3})\]")
    GNB_PATTERNS = {
        "sctp_connect": re.compile(r"SCTP connection established"),
        "sctp_fail": re.compile(r"SCTP could not connect"),
        "ng_setup_success": re.compile(r"NG Setup procedure is successful"),
        "ng_setup_request": re.compile(r"Sending NG Setup Request"),
        "rrc_setup": re.compile(r"RRC Setup for UE"),
        "initial_nas": re.compile(r"Initial NAS message received"),
        "initial_context_setup": re.compile(r"Initial Context Setup Request received"),
        "pdu_setup": re.compile(r"PDU session resource\(s\) setup for UE"),
        "ue_context_release": re.compile(r"UE Context Release Command received"),
        "signal_lost": re.compile(r"signal lost"),
        "signal_detected": re.compile(r"new signal detected"),
    }

    PROXY_SWITCH_RE = re.compile(
        r"\[(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})\].*BACKEND SWITCH: .* -> (\w+)"
    )

    def parse_ue(self, text: str, ue_name: str = "ue") -> List[Dict]:
        events = []
        for line in text.splitlines():
            m = self.UE_TS_RE.search(line)
            if not m:
                continue
            ts = m.group(1)
            for name, pat in self.UE_PATTERNS.items():
                mm = pat.search(line)
                if mm:
                    ev = {"timestamp": ts, "source": ue_name, "event": name, "line": line.strip()}
                    if name == "tun_up" and mm.lastindex:
                        ev["ip"] = mm.group(1)
                    events.append(ev)
                    break
        return events

    def parse_amf(self, text: str, amf_name: str) -> List[Dict]:
        events = []
        for line in text.splitlines():
            m = self.AMF_TS_RE.search(line)
            if not m:
                continue
            ts_raw = m.group(1)
            ts = f"2026-{ts_raw.replace('/', '-')}"
            for name, pat in self.AMF_PATTERNS.items():
                mm = pat.search(line)
                if mm:
                    ev = {"timestamp": ts, "source": amf_name, "event": name, "line": line.strip()}
                    if mm.lastindex and name in ("amf_session_added", "amf_session_removed", "gnb_ue_added", "gnb_ue_removed"):
                        ev["count"] = int(mm.group(1))
                    events.append(ev)
                    break
        return events

    def parse_gnb(self, text: str, gnb_name: str = "gnb1") -> List[Dict]:
        events = []
        for line in text.splitlines():
            m = self.GNB_TS_RE.search(line)
            if not m:
                continue
            ts = m.group(1)
            for name, pat in self.GNB_PATTERNS.items():
                if pat.search(line):
                    events.append({"timestamp": ts, "source": gnb_name, "event": name, "line": line.strip()})
                    break
        return events

    def parse_proxy_switch_time(self, text: str, target_amf: str) -> Optional[str]:
        """Extract the most recent switch timestamp for target_amf from proxy logs."""
        matches = self.PROXY_SWITCH_RE.findall(text)
        if not matches:
            return None
        for ts_str, backend in reversed(matches):
            if backend == target_amf:
                return ts_str
        return None

class MetricComputer:
    """Compute per-UE and aggregate metrics from parsed logs."""

    @staticmethod
    def _ts_to_ms(ts_str: str) -> float:
        s = ts_str.strip()
        for fmt in ("%Y-%m-%d %H:%M:%S.%f", "%Y-%m-%d %H:%M:%S"):
            try:
                dt = datetime.strptime(s, fmt)
                return dt.timestamp() * 1000
            except ValueError:
                continue
        return 0.0

    @classmethod
    def compute_ue_metrics(cls, ue_events: List[Dict], switch_time_ms: float, ue_name: str,
                           restart_time_ms: Optional[float] = None) -> Dict:
        m = {
            "ue": ue_name,
            "switch_time_ms": switch_time_ms,
            "restart_time_ms": restart_time_ms,
            "reallocation_latency_ms": None,
            "registration_latency_ms": None,
            "session_setup_latency_ms": None,
            "rrc_latency_ms": None,
            "post_restart_latency_ms": None,
            "ip_changed": False,
            "old_ip": None,
            "new_ip": None,
            "recovered": False,
            "events": ue_events,
            "registration_success_ms": None,
            "initial_registration_ms": None,
            "tun_up_ms": None,
            "rrc_connected_ms": None,
        }

        sorted_events = sorted(ue_events, key=lambda e: cls._ts_to_ms(e["timestamp"]))
        post_switch = [e for e in sorted_events if cls._ts_to_ms(e["timestamp"]) >= switch_time_ms]

        # Find the LAST registration_success after the switch
        reg_success_idx = None
        for i in range(len(post_switch) - 1, -1, -1):
            if post_switch[i]["event"] == "registration_success":
                reg_success_idx = i
                break

        if reg_success_idx is None:
            return m

        reg_success_event = post_switch[reg_success_idx]
        reg_success_ts = cls._ts_to_ms(reg_success_event["timestamp"])

        # Walk backwards to find initial_registration in the same cycle
        init_reg_event = None
        for i in range(reg_success_idx - 1, -1, -1):
            if post_switch[i]["event"] == "initial_registration":
                init_reg_event = post_switch[i]
                break

        # Walk further back to find mm_deregistered that starts this cycle
        cycle_start_idx = None
        if init_reg_event is not None:
            init_reg_ts = cls._ts_to_ms(init_reg_event["timestamp"])
            for i in range(reg_success_idx - 1, -1, -1):
                if post_switch[i]["event"] == "mm_deregistered":
                    if cls._ts_to_ms(post_switch[i]["timestamp"]) <= init_reg_ts:
                        cycle_start_idx = i
                        break

        # Walk forward to find tun_up after reg_success
        tun_up_event = None
        for i in range(reg_success_idx + 1, len(post_switch)):
            if post_switch[i]["event"] == "tun_up":
                tun_up_event = post_switch[i]
                break

        # Compute metrics
        m["recovered"] = True
        m["reallocation_latency_ms"] = reg_success_ts - switch_time_ms
        m["registration_success_ms"] = reg_success_ts

        if init_reg_event is not None:
            m["registration_latency_ms"] = reg_success_ts - cls._ts_to_ms(init_reg_event["timestamp"])
            m["initial_registration_ms"] = cls._ts_to_ms(init_reg_event["timestamp"])

        if tun_up_event is not None:
            m["session_setup_latency_ms"] = cls._ts_to_ms(tun_up_event["timestamp"]) - reg_success_ts
            m["new_ip"] = tun_up_event.get("ip")
            m["tun_up_ms"] = cls._ts_to_ms(tun_up_event["timestamp"])

        # old_ip: last tun_up BEFORE the cycle-start mm_deregistered
        if cycle_start_idx is not None:
            cycle_start_ts = cls._ts_to_ms(post_switch[cycle_start_idx]["timestamp"])
            pre_cycle = [e for e in sorted_events if cls._ts_to_ms(e["timestamp"]) < cycle_start_ts]
            old_tuns = [e for e in pre_cycle if e["event"] == "tun_up"]
            if old_tuns:
                m["old_ip"] = old_tuns[-1].get("ip")

        if m["old_ip"] and m["new_ip"]:
            m["ip_changed"] = m["old_ip"] != m["new_ip"]

        # RRC latency: rrc_connected between initial_registration and registration_success
        rrc_event = None
        if init_reg_event is not None:
            init_reg_ts = cls._ts_to_ms(init_reg_event["timestamp"])
            for i in range(reg_success_idx - 1, -1, -1):
                if post_switch[i]["event"] == "rrc_connected":
                    ts = cls._ts_to_ms(post_switch[i]["timestamp"])
                    if init_reg_ts <= ts <= reg_success_ts:
                        rrc_event = post_switch[i]
                        break
            if rrc_event:
                m["rrc_latency_ms"] = cls._ts_to_ms(rrc_event["timestamp"]) - init_reg_ts
                m["rrc_connected_ms"] = cls._ts_to_ms(rrc_event["timestamp"])

        # Post-restart latency: from container restart to registration_success
        if restart_time_ms is not None:
            m["post_restart_latency_ms"] = reg_success_ts - restart_time_ms

        return m

    @classmethod
    def compute_aggregate(cls, ue_results: List[Dict]) -> Dict:
        """Compute mean, std, min, max, p95 across UE metrics."""
        def _extract(key: str) -> List[float]:
            return [r[key] for r in ue_results if r.get(key) is not None]

        agg = {}
        for key in ("reallocation_latency_ms", "registration_latency_ms", "session_setup_latency_ms",
                    "rrc_latency_ms", "post_restart_latency_ms"):
            vals = _extract(key)
            if not vals:
                agg[key] = {"mean": None, "std": None, "min": None, "max": None, "p95": None, "count": 0}
                continue
            vals_sorted = sorted(vals)
            p95_idx = int(len(vals_sorted) * 0.95)
            p95 = vals_sorted[min(p95_idx, len(vals_sorted) - 1)]
            agg[key] = {
                "mean": statistics.mean(vals),
                "std": statistics.stdev(vals) if len(vals) > 1 else 0.0,
                "min": min(vals),
                "max": max(vals),
                "p95": p95,
                "count": len(vals),
            }
        agg["recovered_count"] = sum(1 for r in ue_results if r.get("recovered"))
        agg["ip_changed_count"] = sum(1 for r in ue_results if r.get("ip_changed"))
        return agg

    def run_sequential(self, target_amf: str, wait_before: float,
                       wait_after: float, stagger: float = 0.0) -> Dict:
        print(f"\n{'='*60}")
        print(f"Mode: SEQUENTIAL | Target: {target_amf} | UEs: {self.ue_pool}")
        print(f"{'='*60}")

        if not self.run_baseline_check():
            return {"error": "proxy not running"}

        print(f"\n[2/5] Stabilizing for {wait_before}s...")
        time.sleep(wait_before)

        print(f"\n[3/5] Switching proxy to {target_amf}...")
        if not switch_proxy_backend(target_amf, self.proxy_host, self.proxy_port):
            print("[ERROR] Proxy switch failed!")
            return {"error": "switch failed"}

        time.sleep(1.0)
        t0 = self._get_proxy_switch_time_ms(target_amf)
        if t0 is None:
            print("[WARN] Could not parse proxy switch time; falling back to host time")
            t0 = time.time() * 1000
        else:
            print(f"[INFO] Proxy switch detected at {datetime.fromtimestamp(t0/1000).isoformat()}")

        print(f"\n[4/5] Restarting gNB(s) -> UEs sequentially (stagger={stagger}s)...")
        for gnb in self.gnb_pool:
            restart_container(gnb)
        if stagger > 0:
            time.sleep(2)

        results = []
        for ue in self.ue_pool:
            restart_container(ue)
            if stagger > 0:
                time.sleep(stagger)
            res = self.wait_for_ue_recovery(ue, t0, timeout=wait_after)
            results.append(res)

        print(f"\n[5/5] Computing aggregate metrics...")
        agg = self.computer.compute_aggregate(results)
        return {
            "mode": "sequential",
            "target_amf": target_amf,
            "switch_time_ms": t0,
            "ue_results": results,
            "aggregate": agg,
